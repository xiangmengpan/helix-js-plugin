```rust
#[tokio::test(flavor = "multi_thread")]
async fn panel_shrinks_editor() -> anyhow::Result<()> {
    let dir = tempfile::tempdir()?;
    let file = dir.path().join("ly.txt");
    std::fs::write(&file, "x\n")?;
    let plugin_path = dir.path().join("layout.js");
    std::fs::write(
        &plugin_path,
        r#"
        helix.register_command("panel-layout", () => {
            helix.open_panel({ side: "right", size: 20, render: () => ["PANEL-STRIP"] });
        });
        "#,
    )?;

    test_key_sequences(
        &mut AppBuilder::new().with_file(file, None).build()?,
        vec![
            (Some(&format!(":plugin-load {}<ret>", plugin_path.display())), None),
            (
                Some(":panel-layout<ret>"),
                Some(&|app| {
                    // 渲染 compositor 到 Buffer，断言编辑器区域收缩
                    // 底部状态栏由 EditorView 画——收缩后其宽度 = 视口 - 20
                    // 右侧 20 列应为面板内容 "PANEL-STRIP"
                    let area = helix_view::graphics::Rect::new(0, 0, 120, 30);
                    let mut buf = tui::buffer::Buffer::empty(area);
                    let mut cx = helix_term::compositor::Context { editor: &mut app.editor, scroll: None, jobs: &mut app.jobs };
                    app.compositor.render(area, &mut buf, &mut cx);
                    let bottom_row: String = buf.content
                        .iter()
                        .skip(area.y as usize * 120 + (area.height - 1) as usize * 120)
                        .take(120)
                        .map(|c| c.symbol.as_str())
                        .collect();
                    assert!(
                        bottom_row.contains("PANEL-STRIP"),
                        "panel strip should occupy right 20 cols: {bottom_row:?}"
                    );
                    // 状态栏特征（如 mode 指示）不在最右 20 列
                    let right20: String = bottom_row.chars().skip(100).collect();
                    assert!(!right20.contains("NORMAL"), "statusline should not extend under panel: {right20:?}");
                }),
            ),
        ],
        false,
    )
    .await?;

    Ok(())
}
```

> 注意：`app.compositor.render` 需要 `&mut Context`（editor/jobs）——`app.editor`/`app.jobs` 字段可见性以 Application 结构为准（editor 是 pub，jobs 可能 pub）。Buffer 坐标计算（120 宽 × 底部行）以 Rect 布局为准调整。若 render 的 Context 构造在测试里受限，退化为：断言 PluginPanel::split_area 的单测 + compositor render 特化的手工验证，报告中注明。
