#[test]
fn theme_overrides() {
    let _guard = TEST_LOCK.lock().unwrap();
    init();
    load_script(
        r#"
        helix.set_theme({ "ui.popup": "#ff00aa", "error": "red", "bad_color": "not-a-color!!" });
        helix.echo("set:" + take_theme_overrides().unwrap().get("error").unwrap());
