#[test]
fn entry_load_export_lazy() {
    let _guard = TEST_LOCK.lock().unwrap();
    init();
    let dir = tempfile::tempdir().unwrap();  // 若 helix-js 测试没有 tempfile——用 std::env::temp_dir() + 唯一子目录 + fs 清理
    set_plugins_dir(dir.path().to_path_buf());

    // 导出 + 加载往返
    std::fs::write(dir.path().join("exp.js"), r#"helix.export({ a: 1, b: "x" });"#).unwrap();
    load_script_named("init.js", r#"helix.load("exp.js");"#).unwrap();
    // init.js 的 load 本身无法断言返回值——直接测 js_load 路径：
    // 用 helix.run_command 间接：注册命令调用 load 并把结果 echo 出来
    load_script_named(
        "driver.js",
        r#"
        helix.register_command("load-exp", () => {
            const mod = helix.load("exp.js");
            helix.echo("a:" + mod.a + " b:" + mod.b);
        });
        helix.register_command("load-cached", () => {
            const m1 = helix.load("exp.js");
            const m2 = helix.load("exp.js");
            helix.echo("same:" + (m1 === m2));
        });
        "#,
    )
    .unwrap();
    let ctx = CommandContext { path: None, text: String::new(), cursor: (0, 0), selection: ((0, 0), (0, 0)) };
    assert!(run_command("load-exp", &ctx).unwrap());
    assert_eq!(take_messages(), vec!["a:1 b:x"]);
    assert!(run_command("load-cached", &ctx).unwrap());
    assert_eq!(take_messages(), vec!["same:true"]);

    // lazy：桩首次调用时加载 + 转执行
    std::fs::write(
        dir.path().join("lazy.js"),
        r#"helix.register_command("lazy-cmd", () => { helix.echo("lazy-ran"); });"#,
    )
    .unwrap();
    load_script_named("lazy-driver.js", r#"helix.lazy("lazy.js", "lazy-cmd");"#).unwrap();
    assert!(run_command("lazy-cmd", &ctx).unwrap());
    assert_eq!(take_messages(), vec!["lazy-ran"]);

    // run_command 带 ctx：命令读 ctx.cursor
    load_script_named(
        "rc.js",
        r#"
        helix.register_command("where", (c) => { helix.echo("at:" + c.cursor.row + "," + c.cursor.col); });
        "#,
    )
    .unwrap();
    load_script_named(
        "rc-driver.js",
        r#"helix.register_command("call-where", () => { helix.run_command("where", { cursor: { row: 3, col: 7 } }); });"#,
    )
    .unwrap();
    assert!(run_command("call-where", &ctx).unwrap());
    assert_eq!(take_messages(), vec!["at:3,7"]);

    // 校验：未知文件 → 抛错
    load_script_named("bad-driver.js", r#"helix.register_command("bad-load", () => { helix.load("nope.js"); });"#).unwrap();
    assert!(run_command("bad-load", &ctx).is_err());
}
```

    - 缓存：SCRIPT_EXPORTS 有 name → 返回缓存 JsValue
    - 读文件（fs::read_to_string，失败抛 JsError）→ IIFE eval（复用 eval_wrapped）→ `LAST_EXPORT.take().unwrap_or(undefined)` → 记录 LOADED_SCRIPTS（name, src，仅当未记录）→ 返回
  - `js_export`（arity 1）：`LAST_EXPORT = Some(arg)`（undefined/null → None）
  - `js_lazy`（arity ≥2）：name + 命令名们；对每个 cmd：eval 构造桩 `(function(){ return function(ctx){ helix.load(<name-json>); helix.run_command(<cmd-json>, ctx); }; })()` 的调用——或更简单：`eval("(function(n,c){ return function(ctx){ helix.load(n); helix.run_command(c, ctx); }; })")` 后 call(n, c) 得闭包 → REGISTRY.insert(cmd, closure)（与 register_command 同机制）
  - `js_run_command`（arity 1-2）：name + 可选 ctx 对象——从 JS 对象读 path/text/cursor/selection（缺省值）→ CommandContext → `run_command(name, &ctx)` → Err 转 JsError

#[tokio::test(flavor = "multi_thread")]
async fn plugin_entry_import_and_lazy() -> anyhow::Result<()> {
    let dir = tempfile::tempdir()?;
    let file = dir.path().join("e.txt");
    std::fs::write(&file, "x\n")?;
    // 模拟 init.js：导入模块 + 懒加载
    std::fs::write(dir.path().join("mod.js"), r#"helix.register_command("mod-cmd", () => { helix.echo("mod-ok"); });"#)?;
    std::fs::write(dir.path().join("heavy.js"), r#"helix.register_command("heavy-cmd", () => { helix.echo("heavy-ok"); });"#)?;
    let init = dir.path().join("init.js");
    std::fs::write(&init, r#"helix.load("mod.js"); helix.lazy("heavy.js", "heavy-cmd");"#)?;

    test_key_sequences(
        &mut AppBuilder::new().with_file(file, None).build()?,
        vec![
            (Some(&format!(":plugin-load {}<ret>", init.display())), None),
            (Some(":mod-cmd<ret>"), Some(&|app| {
                let (status, _) = app.editor.get_status().unwrap();
                assert_eq!(status.as_ref(), "mod-ok");
            })),
            // 懒加载：首次 heavy-cmd 生效
            (Some(":heavy-cmd<ret>"), Some(&|app| {
                let (status, _) = app.editor.get_status().unwrap();
                assert_eq!(status.as_ref(), "heavy-ok");
            })),
        ],
        false,
    )
    .await?;
    Ok(())
}
```

> 注意：`:plugin-load` 加载的是绝对路径的 init.js——js_load 的相对名解析需要 PLUGINS_DIR 指向 mod.js/heavy.js 所在目录。但 set_plugins_dir 在 Application::new 里设的是真实 plugins 目录！测试里的相对名会解析到真实目录。**方案**：集成测试用绝对路径 `helix.load("<dir>/mod.js")`（js_load 支持绝对路径），绕过目录问题；或 `:plugin-load` 前无法改目录。以绝对路径方案为准（测试 init.js 内容用绝对路径写 mod.js/heavy.js 的完整路径）。
