```rust
fn wait_for_term_event(pred: impl Fn(&TermEvent) -> bool) -> Vec<TermEvent> {
    let deadline = std::time::Instant::now() + std::time::Duration::from_secs(5);
    loop {
        let events = drain_term_events();
        if events.iter().any(&pred) {
            return events;
        }
        if std::time::Instant::now() > deadline {
            panic!("timed out waiting for term event");
        }
        std::thread::sleep(std::time::Duration::from_millis(50));
    }
}
```

```rust
/// 异步进程事件（worker → 主线程）
pub enum TermEvent {
    Chunk(u64, String),
    Exit(u64, i32),
}

enum TermCtrl {
    Write(String),
    Kill,
}

struct TermCallbacks {
    on_chunk: JsValue,
    on_exit: Option<JsValue>,
}

thread_local! {
    static TERM_CALLBACKS: RefCell<Option<&'static mut HashMap<u64, TermCallbacks>>> = const { RefCell::new(None) };
    static NEXT_TERM_ID: Cell<u64> = const { Cell::new(1) };
}
static TERM_EVENTS: OnceLock<std::sync::mpsc::Sender<TermEvent>> = OnceLock::new();
static TERM_EVENTS_RX: OnceLock<Mutex<std::sync::mpsc::Receiver<TermEvent>>> = OnceLock::new();
static TERM_WORKERS: OnceLock<Mutex<HashMap<u64, std::sync::mpsc::Sender<TermCtrl>>>> = OnceLock::new();
```

```rust
const TERM_CHUNK_SIZE: usize = 4096;

/// 启动 worker 线程：跑子进程，逐块读输出发 Chunk，EOF 后 wait 发 Exit；收 stdin/杀进程控制
fn spawn_worker(
    id: u64,
    cmd: &str,
    tx: std::sync::mpsc::Sender<TermEvent>,
    stdin_rx: std::sync::mpsc::Receiver<TermCtrl>,
) {
    let cmd = cmd.to_string();
    std::thread::spawn(move || {
        use std::io::Read;
        use std::io::Write;
        use std::process::{Command, Stdio};
        let mut child = match Command::new("sh")
            .arg("-c")
            .arg(&cmd)
            .stdin(Stdio::piped())
            .stdout(Stdio::piped())
            .stderr(Stdio::piped())
            .spawn()
        {
            Ok(c) => c,
            Err(e) => {
                let _ = tx.send(TermEvent::Exit(id, 127));
                return;
            }
        };
        let mut stdout = child.stdout.take();
        let mut stderr = child.stderr.take();
        let mut stdin = child.stdin.take();

        // 读输出线程（合并 stdout+stderr，逐块）
        let tx_read = tx.clone();
        let mut handles = Vec::new();
        for mut stream in [stdout, stderr].into_iter().flatten() {
            let tx = tx_read.clone();
            handles.push(std::thread::spawn(move || {
                let mut buf = [0u8; TERM_CHUNK_SIZE];
                loop {
                    match stream.read(&mut buf) {
                        Ok(0) => break,
                        Ok(n) => {
                            let chunk = String::from_utf8_lossy(&buf[..n]).to_string();
                            if tx.send(TermEvent::Chunk(id, chunk)).is_err() {
                                break;
                            }
                        }
                        Err(_) => break,
                    }
                }
            }));
        }

        // 控制循环：写 stdin / kill
        let kill_flag = std::sync::Arc::new(std::sync::atomic::AtomicBool::new(false));
        {
            let kill_flag = kill_flag.clone();
            let mut stdin_opt = stdin;
            let stdin_t = std::thread::spawn(move || {
                for msg in stdin_rx {
                    match msg {
                        TermCtrl::Write(text) => {
                            if let Some(s) = stdin_opt.as_mut() {
                                let _ = s.write_all(text.as_bytes());
                                let _ = s.flush();
                            }
                        }
                        TermCtrl::Kill => {
                            kill_flag.store(true, std::sync::atomic::Ordering::SeqCst);
                            break;
                        }
                    }
                }
            });
            let _ = stdin_t.join();
        }

        for h in handles {
            let _ = h.join();
        }
        let status = if kill_flag.load(std::sync::atomic::Ordering::SeqCst) {
            let _ = child.kill();
            child.wait()
        } else {
            let _ = stdin.take();
            child.wait()
        };
        let code = status.map(|s| s.code().unwrap_or(-1)).unwrap_or(-1);
        let _ = tx.send(TermEvent::Exit(id, code));
    });
}
```

> 注意：stdout/stderr 两个读线程 + 控制线程 + 主 worker 线程的结构以"能编译、行为正确"为准——实现者可简化（如单线程非阻塞 select 不可行，用两个读线程是合理折中）。`child.stdin.take()` 的 Option 移动问题（控制线程持有 Option<ChildStdin>，结束后 drop 让子进程 EOF）以编译为准调整。Exit code 对 signal 终止传 `-signal` 或 128+signal——测试只断言非 0，以实现为准并在报告注明。

```rust
fn js_run_async(_this: &JsValue, args: &[JsValue], context: &mut Context) -> boa_engine::JsResult<JsValue> {
    let cmd: String = args.first().unwrap_or(&JsValue::undefined()).try_js_into(context)?;
    let cb = args.get(1).cloned().unwrap_or(JsValue::undefined());
    if cb.as_callable().is_none() {
        return Err(JsError::from_opaque(JsValue::from(JsString::from("helix.run_async: callback must be a function"))));
    }
    let id = NEXT_TERM_ID.with(|c| { let v = c.get(); c.set(v + 1); v });
    TERM_CALLBACKS.with(|t| {
        let slot = t.borrow_mut();
        let map = slot.get_or_insert_with(|| Box::leak(Box::default()));
        map.insert(id, TermCallbacks { on_chunk: cb.clone(), on_exit: Some(cb) });
    });
    // run_async 复用 spawn worker；Exit 事件时回调 (err, out)——stdout 收集由 worker 的 Chunk 事件提供？
    // —— run_async 语义：完成回调拿到完整 stdout。两种实现：
    //   a) 单独跑一次性命令（不流式），Exit 时带 stdout——TermEvent::Exit 只带 code，不带 out → 需新事件或扩展
    //   b) run_async = spawn + 内部聚合 Chunk 直到 Exit，再调回调 (err, aggregated_out)
    // 采用 (b)：resolve_term_event 对 run_async 注册的回调在收到 Exit 时，把该 id 聚合的 stdout 拼好传入。
    // —— 但聚合状态在哪？worker 线程聚合再在 Exit 事件里带出来更简单：TermEvent::Exit(u64, i32, String /*stdout*/)？
    // 让实现者选择：要么 TermEvent::Exit 扩展带 stdout（run_async 专用），要么主线程聚合。
    // 建议：扩展 TermEvent::Exit(u64, i32, Option<String>)——run_async 的 worker 聚合 stdout 放这里；spawn 的 worker 传 None。
    // 简化方案（推荐）：run_async 的 worker 聚合 stdout 于 Exit；spawn 的 worker 流式 Chunk + Exit(None)。
    // 实现时以这个方向为准，测试断言配套调整。
    Ok(JsValue::from(id))
}
```

> **重要决策点**：run_async 的 stdout 聚合。推荐：worker 函数加参数 `aggregate: bool`——true 时把读到的 Chunk 内容聚合进一个 String，Exit 事件携带完整 stdout。`TermEvent::Exit(u64, i32, Option<String>)`（spawn 的 Exit 传 None）。测试断言据此调整。这是规格的"以实测为准"点，实现者选定后在报告注明。

```rust
pub fn drain_term_events() -> Vec<TermEvent> {
    init();
    let rx = TERM_EVENTS_RX.get().expect("TERM_EVENTS_RX initialized");
    let mut events = Vec::new();
    while let Ok(e) = rx.lock().unwrap().try_recv() {
        events.push(e);
    }
    events
}

pub fn resolve_term_event(id: u64, event: TermEvent) -> Result<()> {
    init();
    CONTEXT.with(|cell| {
        let mut binding = cell.borrow_mut();
        let engine = binding.as_mut().expect("CONTEXT initialized");
        let callbacks = TERM_CALLBACKS.with(|t| {
            let slot = t.borrow_mut();
            slot.get_or_insert_with(|| Box::leak(Box::default())).get(&id).map(|c| TermCallbacks {
                on_chunk: c.on_chunk.clone(),
                on_exit: c.on_exit.clone(),
            })
        });
        let Some(callbacks) = callbacks else { return Ok(()) };
        let undefined = JsValue::undefined();
        match event {
            TermEvent::Chunk(_, chunk) => {
                let func = callbacks.on_chunk.as_callable().and_then(JsFunction::from_object)
                    .ok_or_else(|| anyhow!("term {id} onChunk not callable"))?;
                let _: JsValue = func.call(&undefined, &[JsValue::from(JsString::from(chunk))], engine)
                    .map_err(|e| anyhow!("term {id} onChunk failed: {e}"))?;
            }
            TermEvent::Exit(_, code, stdout) => {
                if let Some(on_exit) = callbacks.on_exit {
                    let func = on_exit.as_callable().and_then(JsFunction::from_object)
                        .ok_or_else(|| anyhow!("term {id} onExit not callable"))?;
                    // run_async 语义：(err, out)；spawn 语义：(code)
                    // —— 通过 on_exit 与 on_chunk 是否同一引用区分：run_async 存了同一 cb 到两个字段
                    // 或增加 TermCallbacks 标志。以实现为准：run_async 的 Exit 调 cb(null, stdout)；
                    // spawn 的 Exit 调 cb(code)。
                }
            }
        }
        Ok(())
    })
}
```

> **决策点**：`resolve_term_event` 区分 run_async vs spawn 的 Exit 回调语义（run_async: `(err, out)`；spawn: `(code)`）。实现方案任选（TermCallbacks 加 `run_async: bool` 标志，或 on_exit 存包装闭包——JS 里包一层 `(code) => cb(code===0?null:"exit "+code, stdout?)` 不现实因为 stdout 在 Rust 侧）。推荐：`TermCallbacks { on_chunk, on_exit, is_run_async: bool }`，resolve 时按标志调 `(null, stdout)` 或 `(code)`。选定后在报告注明。
