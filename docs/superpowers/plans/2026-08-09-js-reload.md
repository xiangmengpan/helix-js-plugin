```rust
#[test]
fn plugin_reload() {
    let _guard = TEST_LOCK.lock().unwrap();
    init();
    load_script_named("a.js", r#"helix.register_command("reload-cmd", () => { helix.echo("v1"); });"#).unwrap();
    load_script_named("b.js", r#"helix.on("save", () => {});"#).unwrap();

    assert!(has_handlers("save"));
    let ctx = CommandContext { path: None, text: String::new(), cursor: (0, 0) };
    assert!(run_command("reload-cmd", &ctx).unwrap());
    assert_eq!(take_messages(), vec!["v1"]);

    // reload：清空状态后重跑
    reload_all().unwrap();
    assert!(has_handlers("save"), "handlers re-registered after reload");
    assert!(run_command("reload-cmd", &ctx).unwrap());
    assert_eq!(take_messages(), vec!["v1"]);

    // 失败脚本：reload 报错且含脚本名
    load_script_named("bad.js", r#"throw new Error("boom");"#).is_err(); // 加载时即失败？不——load 时 throw 会 Err
    // 若 load 时 throw 已 Err，bad.js 未入 LOADED_SCRIPTS；为测 reload 失败路径，用"加载成功但 reload 时失败"的脚本不可行（脚本本身是静态的）——
    // 改为：注册一个 reload 后仍失败的场景——脚本在顶层抛错会在 load 时失败。因此 reload 失败路径的测试改为：
    // 直接调用 reload_all 时若某个记录脚本抛错才 Err；用正常脚本时 reload 必然成功。
    // 结论：失败路径无法用顶层抛错脚本触发（load 时就拦了）。可接受——reload_all 的 Err 分支靠代码审查保证，单测只覆盖成功路径。
}
```

> 注意：顶层抛错的脚本在 load 时就会 Err、不会进入 LOADED_SCRIPTS——所以 reload 的失败路径不可由"坏脚本"触发。若想覆盖失败路径，可加一个 `helix.echo` 触发失败？（echo 不失败）。接受：单测覆盖成功路径；失败路径（recorded script 重跑抛错——仅当脚本依赖被清空的状态时才发生，如脚本注册命令后 reload 时 REGISTRY 已清——不会，重跑会重新注册）。**失败路径实际不可达**（记录的都是成功加载过的脚本，重跑同一源码必然再次成功，除非全局状态被清空导致依赖缺失——脚本不依赖其他脚本的状态则不会）。在报告与代码注释中说明这一分析，不为不可达路径写测试。

`load_script` 改为包装 + 新增 named 版本：

```rust
/// 求值一段插件脚本（无名字，用于测试/内联）
pub fn load_script(src: &str) -> Result<()> {
    load_script_named("<anon>", src)
}

/// 求值并记录脚本（名字用于报错定位与热重载）
pub fn load_script_named(name: &str, src: &str) -> Result<()> {
    init();
    CONTEXT.with(|cell| {
        let mut binding = cell.borrow_mut();
        let engine = binding.as_mut().expect("CONTEXT initialized");
        engine
            .eval(Source::from_bytes(src))
            .map(|_| ())
            .map_err(|e| anyhow!("plugin script '{name}' error: {e}"))?;
        LOADED_SCRIPTS.with(|s| s.borrow_mut().push((name.to_string(), src.to_string())));
        Ok(())
    })
}
```

> 注意：既有 load_script 的错误消息是 "plugin script error: {e}"——改 named 后带名字，若有测试断言旧消息需同步（检查既有测试）。

reset 辅助 + reload_all：

```rust
/// 清空所有插件状态（热重载用；id 计数器保留保证唯一性）
fn reset_plugin_state() {
    REGISTRY.with(|r| r.borrow_mut().clear());
    EVENT_HANDLERS.with(|h| h.borrow_mut().clear());
    POPUPS.with(|p| p.borrow_mut().clear());
    CURRENT_EDITS.with(|c| c.borrow_mut().clear());
    BUFFER_ICON_HOOK.with(|b| *b.borrow_mut() = None);
    STATUSLINE_HOOK.with(|s| *s.borrow_mut() = None);
    COMMAND_DOCS.with(|d| d.borrow_mut().clear());
}

/// 热重载：清空状态后按加载顺序重跑全部脚本
pub fn reload_all() -> Result<()> {
    init();
    let scripts = LOADED_SCRIPTS.with(|s| s.borrow().clone());
    reset_plugin_state();
    for (name, src) in &scripts {
        CONTEXT.with(|cell| {
            let mut binding = cell.borrow_mut();
            let engine = binding.as_mut().expect("CONTEXT initialized");
            engine
                .eval(Source::from_bytes(src.as_str()))
                .map(|_| ())
                .map_err(|e| anyhow!("plugin reload '{name}' failed: {e}"))?;
            Ok(())
        })?;
    }
    Ok(())
}
```

> 注意：`reset_plugin_state` 引用的状态名（REGISTRY/EVENT_HANDLERS/POPUPS/...）以 helix-js 实际代码为准——D/C 合并后 COMMAND_DOCS 存在。`STATUSLINE_HOOK` 若在（特性 B 已合并）则在。所有名字对照当前 lib.rs 核实，缺失的跳过。UI_REQUESTS/MESSAGES 不清（它们是命令边界 drain 的队列，清了也无害——可清可不清，以简洁为准）。

函数：

```rust
fn plugin_reload(cx: &mut compositor::Context, _args: Args, event: PromptEvent) -> anyhow::Result<()> {
    if event != PromptEvent::Validate {
        return Ok(());
    }
    match helix_js::reload_all() {
        Ok(()) => {
            let msgs = helix_js::take_messages();
            if !msgs.is_empty() {
                cx.editor.set_status(msgs.join(" "));
            }
            Ok(())
        }
        Err(err) => {
            cx.editor.set_error(format!("plugin-reload: {err}"));
            Ok(())
        }
    }
}
```

- application.rs 启动循环：`helix_js::load_script(&src)` → `helix_js::load_script_named(&path.display().to_string(), &src)`

```rust
use super::*;

#[tokio::test(flavor = "multi_thread")]
async fn plugin_reload_command() -> anyhow::Result<()> {
    let dir = tempfile::tempdir()?;
    let file = dir.path().join("r.txt");
    std::fs::write(&file, "x\n")?;
    let plugin_path = dir.path().join("reload.js");
    std::fs::write(
        &plugin_path,
        r#"helix.register_command("reload-demo", () => { helix.echo("reloaded-ok"); });"#,
    )?;

    test_key_sequences(
        &mut AppBuilder::new().with_file(file, None).build()?,
        vec![
            (Some(&format!(":plugin-load {}<ret>", plugin_path.display())), None),
            (Some(":reload-demo<ret>"), Some(&|app| {
                let (status, _) = app.editor.get_status().unwrap();
                assert_eq!(status.as_ref(), "reloaded-ok");
            })),
            (Some(":plugin-reload<ret>"), Some(&|app| {
                // reload 后命令仍可用（重跑注册）
            })),
            (Some(":reload-demo<ret>"), Some(&|app| {
                let (status, severity) = app.editor.get_status().unwrap();
                assert_eq!(status.as_ref(), "reloaded-ok");
                assert!(matches!(severity, helix_core::diagnostic::Severity::Info), "no error on reload");
            })),
        ],
        false,
    )
    .await?;

    Ok(())
}
```
