```rust
fn js_open_popup(_this: &JsValue, args: &[JsValue], ctx: &mut Context) -> boa_engine::JsResult<JsValue> {
    let opts = args.first().unwrap_or(&JsValue::undefined()).as_object().ok_or_else(|| {
        JsError::from_opaque(JsValue::from("open_popup: options object required"))
    })?;
    let render = opts.get("render", ctx)?;
    if render.as_callable().is_none() {
        return Err(JsError::from_opaque(JsValue::from("open_popup: render must be a function")));
    }
    let on_key = opts.get("onKey", ctx)?;
    let on_key = on_key.as_callable().map(|_| on_key);
    let on_close = opts.get("onClose", ctx)?;
    let on_close = on_close.as_callable().map(|_| on_close);

    let id = NEXT_POPUP_ID.with(|c| { let v = c.get(); c.set(v + 1); v });
    POPUPS.with(|p| p.borrow_mut().insert(id, PopupCallbacks { render, on_key, on_close }));
    UI_REQUESTS.get().unwrap().lock().unwrap().push(UiRequest::OpenPopup { id });
    Ok(JsValue::from(id))
}

fn js_set_buffer_icon(_this: &JsValue, args: &[JsValue], ctx: &mut Context) -> boa_engine::JsResult<JsValue> {
    let hook = args.first().unwrap_or(&JsValue::undefined());
    if !hook.is_callable() {
        return Err(JsError::from_opaque(JsValue::from("set_buffer_icon: expected a function")));
    }
    BUFFER_ICON_HOOK.with(|h| *h.borrow_mut() = Some(hook.clone()));
    Ok(JsValue::undefined())
}

pub fn take_ui_requests() -> Vec<UiRequest> {
    init();
    std::mem::take(&mut *UI_REQUESTS.get().expect("UI_REQUESTS initialized").lock().expect("ui requests lock"))
}

/// 调 JS render 回调，返回行数组。ctx 对象 { width, height }。
pub fn render_popup(id: u64, width: u16, height: u16) -> Result<Vec<String>> {
    init();
    CONTEXT.with(|cell| {
        let mut binding = cell.borrow_mut();
        let engine = binding.as_mut().expect("CONTEXT initialized");
        let render = POPUPS.with(|p| p.borrow().get(&id).map(|cb| cb.render.clone()))
            .ok_or_else(|| anyhow!("popup {id} not open"))?;
        let ctx_obj = ObjectInitializer::new(engine)
            .property("width", width, Attribute::all())
            .property("height", height, Attribute::all())
            .build();
        let func = render.as_callable().and_then(JsFunction::from_object)
            .ok_or_else(|| anyhow!("popup {id} render is not a function"))?;
        let value: JsValue = func.call(engine, (ctx_obj,))
            .map_err(|e| anyhow!("popup {id} render failed: {e}"))?;
        // 结果必须是 string[]
        let arr: boa_engine::object::builtins::JsArray = value.try_js_into(engine)
            .map_err(|e| anyhow!("popup {id} render must return an array of strings: {e}"))?;
        let len: usize = arr.get("length", engine)?.try_js_into(engine)?;
        let mut lines = Vec::with_capacity(len);
        for i in 0..len {
            let item = arr.get(i, engine)?;
            let s: String = item.try_js_into(engine)
                .map_err(|e| anyhow!("popup {id} render line {i} must be a string: {e}"))?;
            lines.push(s);
        }
        Ok(lines)
    })
}

/// 调 JS onKey 回调。未注册 onKey 或缺省时：Esc→Close，其他→Ignore。
pub fn popup_key(id: u64, key: &PluginKey) -> Result<PopupKeyResult> {
    init();
    CONTEXT.with(|cell| {
        let mut binding = cell.borrow_mut();
        let engine = binding.as_mut().expect("CONTEXT initialized");
        let callbacks = POPUPS.with(|p| p.borrow().get(&id).cloned())
            .ok_or_else(|| anyhow!("popup {id} not open"))?;
        let Some(on_key) = callbacks.on_key else {
            return Ok(if key.name == "Esc" { PopupKeyResult::Close } else { PopupKeyResult::Ignored });
        };
        let func = on_key.as_callable().and_then(JsFunction::from_object)
            .ok_or_else(|| anyhow!("popup {id} onKey is not a function"))?;
        let key_obj = ObjectInitializer::new(engine)
            .property("name", key.name.clone(), Attribute::all())
            .property("shift", key.shift, Attribute::all())
            .property("ctrl", key.ctrl, Attribute::all())
            .property("alt", key.alt, Attribute::all())
            .build();
        let value: JsValue = func.call(engine, (key_obj,))
            .map_err(|e| anyhow!("popup {id} onKey failed: {e}"))?;
        let s: Option<String> = value.try_js_into(engine).ok();
        Ok(match s.as_deref() {
            Some("close") => PopupKeyResult::Close,
            Some("handled") => PopupKeyResult::Handled,
            Some("ignore") => PopupKeyResult::Ignored,
            _ => PopupKeyResult::Handled, // 未识别返回值 → 消费（安全默认）
        })
    })
}

/// 关闭弹窗：触发 onClose 并移除注册表项。幂等（已关闭返回 Ok）。
pub fn close_popup(id: u64) -> Result<()> {
    init();
    CONTEXT.with(|cell| {
        let mut binding = cell.borrow_mut();
        let engine = binding.as_mut().expect("CONTEXT initialized");
        let callbacks = POPUPS.with(|p| p.borrow_mut().remove(&id));
        let Some(callbacks) = callbacks else { return Ok(()) };
        if let Some(on_close) = callbacks.on_close {
            let func = on_close.as_callable().and_then(JsFunction::from_object)
                .ok_or_else(|| anyhow!("popup {id} onClose is not a function"))?;
            let _: JsValue = func.call(engine, ()).map_err(|e| anyhow!("popup {id} onClose failed: {e}"))?;
        }
        Ok(())
    })
}

/// 调 bufferline 图标钩子；未注册 / 返回 null / 报错 → None。
pub fn bufferline_icon(path: Option<&str>) -> Option<String> {
    init();
    CONTEXT.with(|cell| {
        let mut binding = cell.borrow_mut();
        let engine = binding.as_mut().expect("CONTEXT initialized");
        let hook = BUFFER_ICON_HOOK.with(|h| h.borrow().clone());
        let hook = hook?;
        let func = hook.as_callable().and_then(JsFunction::from_object)?;
        let arg = match path { Some(p) => JsValue::from(p.to_string()), None => JsValue::null() };
        let value: JsValue = func.call(engine, (arg,)).ok()?;
        value.try_js_into::<String>(engine).ok()
    })
}
```

> 注意：`PopupCallbacks` 需要 `Clone`（`popup_key` 里 `p.borrow().get(&id).cloned()`）——`#[derive(Clone)]`。`JsValue` 是 Clone 的。
> `func.call(engine, ())` 空参元组需要 `TryIntoJsArguments` 的 0 元组实现（`impl_try_into_js_args!` 宏在 boa 中从 1 元组开始，若 0 元组不可用，改用 `func.call(engine, (JsValue::undefined(),))` 并让 JS 忽略多余参数）。
> 若 `JsArray::get` 的 `length`/`i` 键类型有出入，用 `boa_engine::property::PropertyKey::from(...)` 或 `usize`（JsObject::get 接受 `impl Into<PropertyKey>`，数字/字符串均可用）。

```rust
use crate::compositor::{Component, Context, Event, EventResult};
use tui::buffer::Buffer as Surface;
use tui::text::Text as TuiText;
use tui::widgets::{Paragraph, Widget, Wrap};

use helix_view::graphics::Rect;
use helix_view::input::{Event as InputEvent, KeyCode, KeyEvent, KeyModifiers};
use helix_js::{PluginKey, PopupKeyResult};

/// 由 JS 渲染内容并处理按键的弹窗内容组件（作为 ui::Popup 的子组件使用）。
pub struct PluginPopup {
    id: u64,
    lines: Vec<String>,
}

impl PluginPopup {
    pub fn new(id: u64) -> Self {
        Self { id, lines: Vec::new() }
    }
}

impl Component for PluginPopup {
    fn render(&mut self, area: Rect, surface: &mut Surface, _cx: &mut Context) {
        match helix_js::render_popup(self.id, area.width, area.height) {
            Ok(lines) => self.lines = lines,
            Err(err) => {
                self.lines = vec![format!("<plugin popup error: {err}>")];
            }
        }
        let text = TuiText::from(self.lines.join("\n"));
        let par = Paragraph::new(&text).wrap(Wrap { trim: false });
        par.render(area, surface);
    }

    fn required_size(&mut self, viewport: (u16, u16)) -> Option<(u16, u16)> {
        let width = self.lines.iter().map(|l| l.chars().count() as u16).max().unwrap_or(0).min(viewport.0);
        let height = self.lines.len() as u16;
        Some((width, height))
    }

    fn handle_event(&mut self, event: &Event, _ctx: &mut Context) -> EventResult {
        let Event::Key(key_event) = event else {
            return EventResult::Ignored(None);
        };
        if key_event.kind != KeyEventKind::Press {
            return EventResult::Ignored(None);
        }
        let Some(key) = key_to_plugin_key(key_event) else {
            return EventResult::Ignored(None);
        };
        match helix_js::popup_key(self.id, &key) {
            Ok(PopupKeyResult::Close) => {
                let id = self.id;
                EventResult::Consumed(Some(Callback::EditorCompositor(Box::new(
                    move |_editor: &mut Editor, compositor: &mut Compositor| {
                        let _ = helix_js::close_popup(id);
                        compositor.pop();
                    },
                ))))
            }
            Ok(PopupKeyResult::Handled) => EventResult::Consumed(None),
            Ok(PopupKeyResult::Ignored) | Err(_) => EventResult::Ignored(None),
        }
    }
}

fn key_to_plugin_key(key: &KeyEvent) -> Option<PluginKey> {
    let name = match key.code {
        KeyCode::Char(c) => c.to_string(),
        KeyCode::Enter => "Enter".into(),
        KeyCode::Esc => "Esc".into(),
        KeyCode::Backspace => "Backspace".into(),
        KeyCode::Tab => "Tab".into(),
        KeyCode::BackTab => "BackTab".into(),
        KeyCode::Up => "Up".into(),
        KeyCode::Down => "Down".into(),
        KeyCode::Left => "Left".into(),
        KeyCode::Right => "Right".into(),
        KeyCode::Home => "Home".into(),
        KeyCode::End => "End".into(),
        KeyCode::PageUp => "PageUp".into(),
        KeyCode::PageDown => "PageDown".into(),
        KeyCode::Delete => "Delete".into(),
        KeyCode::Insert => "Insert".into(),
        KeyCode::F(n) => format!("F{n}"),
        _ => return None,
    };
    Some(PluginKey {
        name,
        shift: key.modifiers.contains(KeyModifiers::SHIFT),
        ctrl: key.modifiers.contains(KeyModifiers::CONTROL),
        alt: key.modifiers.contains(KeyModifiers::ALT),
    })
}
```

> 注意：`KeyEventKind`（Press/Repeat/Release）与 `KeyEvent`、`KeyCode`、`KeyModifiers` 的导入路径以编译器为准（`helix_view::input` 或 `helix_view::keyboard`）；`Callback`、`Compositor`、`Editor` 从 `crate::compositor` / `helix_view::Editor` 导入（参照 `ui/editor.rs` 既有模式）。`Popup` 的 auto_close 也会处理 Esc——注意 PluginPopup 已消费 Esc（onKey 返回 close），若 auto_close 与 JS 处理冲突，外层 Popup 用 `.auto_close(false)` 或让 PluginPopup 的 Esc 走 Ignored 由外层关（实现时以集成测试结果为准，保持行为：Esc 关闭弹窗且触发 onClose）。
