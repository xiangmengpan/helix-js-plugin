```rust
#[test]
fn pty_spawn() {
    let _guard = TEST_LOCK.lock().unwrap();
    init();
    load_script(
        r#"
        helix.register_command("pty-tty", () => {
            const id = helix.spawn({ pty: true, cmd: "tty", onChunk: (c) => helix.echo("out:" + c.trim()), onExit: (code) => helix.echo("exit:" + code) });
        });
        helix.register_command("pty-size", () => {
            const id = helix.spawn({ pty: true, cmd: "stty size", onChunk: (c) => helix.echo("size:" + c.trim()) });
        });
        "#,
    )
    .unwrap();
    let ctx = CommandContext { path: None, text: String::new(), cursor: (0, 0), selection: ((0, 0), (0, 0)) };
    assert!(run_command("pty-tty", &ctx).unwrap());
    let events = wait_for_term_event(|e| matches!(e, TermEvent::Chunk(_, c) if c.contains("/dev/pts/")));
    assert!(true, "tty command sees a pty");
    // resolve 并断言 echo
    if let TermEvent::Chunk(id, c) = events[0] { resolve_term_event(id, TermEvent::Chunk(id, c.clone())).unwrap(); }
    assert!(take_messages().iter().any(|m| m.contains("/dev/pts/")));
}
```

> 注意：pty 输出的行缓冲——`tty` 在 PTY 下输出带 `\r\n`（终端换行）而非 `\n`；chunk 断言用 contains 而非精确匹配。`stty size` 断言 "24 80"（默认 winsize）。term_resize 测试：spawn `stty size` 前先 term_resize(id, 40, 100)——但 id 在 JS 内无法先拿到再 resize（spawn 返回 id 后 resize 需在 onChunk 前完成——时序问题！）。方案：resize 测试用两条命令链——命令里 spawn 后立即 term_resize 再 echo，chunk 到 "100 40"。行列映射（stty 输出 "rows cols"）以实测为准。
